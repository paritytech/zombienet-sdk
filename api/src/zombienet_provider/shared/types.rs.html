28/// Role of a node in the network, exposed by providers that support labels
29/// (k8s) as the `zombie-role` label with the values zombienet v1 used
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub enum NodeRole {
32    /// Relay chain node, validator or not
33    Node,
34    /// Dedicated bootnode
35    Bootnode,
36    /// Collator of a parachain that is not cumulus based (e.g adder/undying)
37    Collator,
38    /// Collator of a cumulus based parachain
39    CumulusCollator,
40    /// Short-lived helper node (e.g used to generate chain-specs)
41    Temp,
42    /// A custom process declared in the network config, running next to the nodes
43    CustomProcess,
44}
45
46impl NodeRole {
47    pub fn as_label(&self) -> &'static str {
48        match self {
49            NodeRole::Node => "node",
50            NodeRole::Bootnode => "bootnode",
51            NodeRole::Collator => "collator",
52            NodeRole::CumulusCollator => "cumulus-collator",
53            NodeRole::Temp => "temp",
54            NodeRole::CustomProcess => "custom-process",
55        }
56    }
57}

59#[derive(Debug, Clone)]
60pub struct SpawnNodeOptions {
61    /// Name of the node
62    pub name: String,
63    /// Image of the node (IFF is supported by the provider)
64    pub image: Option<String>,
65    /// Resources to apply to the node (IFF is supported by the provider)
66    pub resources: Option<Resources>,
67    /// Main command to execute
68    pub program: String,
69    /// Arguments to pass to the main command
70    pub args: Vec<String>,
71    /// Environment to set when running the `program`
72    pub env: Vec<(String, String)>,
73    // TODO: rename startup_files
74    /// Files to inject at startup
75    pub injected_files: Vec<TransferedFile>,
76    /// Paths to create before start the node (e.g keystore)
77    /// should be created with `create_dir_all` in order
78    /// to create the full path even when we have missing parts
79    pub created_paths: Vec<PathBuf>,
80    /// Database snapshot to be injected (a local `.tgz` resolved by the
81    /// orchestrator before spawn — providers extract it directly, no
82    /// download). See `orchestrator::generators::resolve_db_snapshots`.
83    pub db_snapshot: Option<PathBuf>,
84    pub port_mapping: Option<HashMap<Port, Port>>,
85    /// Named ports the process listens on. On k8s they are the pod's container
86    /// ports and its `Service` ports (none gives a headless Service), and the
87    /// first is the readiness probe's target when `wrapper` is off; the other
88    /// providers ignore them.
89    pub ports: Vec<(String, Port)>,
90    /// Optionally specify a log path for the node
91    pub node_log_path: Option<PathBuf>,
92    /// Role of the node in the network
93    pub role: Option<NodeRole>,
94    /// Run `program` through zombienet's wrapper script, which gives the
95    /// pause/resume/restart controls nodes have. The wrapper needs `bash` in
96    /// the image and replaces the image's entrypoint, so a custom process that
97    /// is just an off-the-shelf image runs without it.
98    pub wrapper: bool,
99}

116            port_mapping: None,
117            ports: vec![],
118            node_log_path: None,
119            role: None,
120            wrapper: true,
121        }
122    }
123
124    pub fn image<S>(mut self, image: S) -> Self
125    where
126        S: AsRef<str>,
127    {
128        self.image = Some(image.as_ref().to_string());
129        self
130    }
131
132    pub fn resources(mut self, resources: Resources) -> Self {
133        self.resources = Some(resources);
134        self
135    }
136
137    pub fn db_snapshot(mut self, db_snap: Option<PathBuf>) -> Self {
138        self.db_snapshot = db_snap;

142    pub fn args<S, I>(mut self, args: I) -> Self
143    where
144        S: AsRef<str>,
145        I: IntoIterator<Item = S>,
146    {
147        self.args = args.into_iter().map(|s| s.as_ref().to_string()).collect();
148        self
149    }
150
151    pub fn env<S, I>(mut self, env: I) -> Self
152    where
153        S: AsRef<str>,
154        I: IntoIterator<Item = (S, S)>,
155    {
156        self.env = env
157            .into_iter()
158            .map(|(name, value)| (name.as_ref().to_string(), value.as_ref().to_string()))
159            .collect();
160        self
161    }
162
163    pub fn injected_files<I>(mut self, injected_files: I) -> Self
164    where
165        I: IntoIterator<Item = TransferedFile>,
166    {
167        self.injected_files = injected_files.into_iter().collect();
168        self
169    }
170
171    pub fn created_paths<P, I>(mut self, created_paths: I) -> Self
172    where
173        P: AsRef<Path>,
174        I: IntoIterator<Item = P>,
175    {
176        self.created_paths = created_paths
177            .into_iter()
178            .map(|path| path.as_ref().into())
179            .collect();
180        self
181    }
182
183    pub fn port_mapping(mut self, ports: HashMap<Port, Port>) -> Self {
184        self.port_mapping = Some(ports);
185        self
186    }
187
188    pub fn node_log_path(mut self, path: Option<PathBuf>) -> Self {
189        self.node_log_path = path;
190        self
191    }
192
193    pub fn role(mut self, role: NodeRole) -> Self {
194        self.role = Some(role);
195        self
196    }
197
198    pub fn ports<S, I>(mut self, ports: I) -> Self
199    where
200        S: AsRef<str>,
201        I: IntoIterator<Item = (S, Port)>,
202    {
203        self.ports = ports
204            .into_iter()
205            .map(|(name, port)| (name.as_ref().to_string(), port))
206            .collect();
207        self
208    }
209
210    /// Run `program` directly instead of through the zombie wrapper script.
211    ///
212    /// The wrapper needs `bash` in the image and is what pause, resume and
213    /// restart talk to, so those return an error for a node spawned this way
214    /// on docker and kubernetes (native signals the process and needs no
215    /// wrapper). The wrapper also holds the program back until the injected
216    /// files are copied in; without it the program starts at once, so it must
217    /// not need any `injected_files` at startup.
218    pub fn without_wrapper(mut self) -> Self {
219        self.wrapper = false;
220        self
221    }
222}

414#[derive(Debug, Clone, Serialize, Deserialize)]
415pub struct TransferedFile {
416    pub local_path: PathBuf,
417    pub remote_path: PathBuf,
418    // TODO: Can be narrowed to have strict typing on this?
419    pub mode: String,
420}
421
422impl TransferedFile {
423    pub fn new<P>(local_path: P, remote_path: P) -> Self
424    where
425        P: AsRef<Path>,
426    {
427        Self {
428            local_path: local_path.as_ref().into(),
429            remote_path: remote_path.as_ref().into(),
430            mode: "0644".to_string(), // default to rw-r--r--
431        }
432    }
433
434    pub fn mode<S>(mut self, mode: S) -> Self
435    where
436        S: AsRef<str>,
437    {
438        self.mode = mode.as_ref().to_string();
439        self
440    }
441}
442
443impl std::fmt::Display for TransferedFile {
444    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
445        write!(
446            f,
447            "File to transfer (local: {}, remote: {})",
448            self.local_path.display(),
449            self.remote_path.display()
450        )
451    }
452}
