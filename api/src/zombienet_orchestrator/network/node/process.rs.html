1//! Custom processes: the services a network config declares under
2//! `[[custom_processes]]`, running next to the nodes.
3//!
4//! A [`CustomProcessNode`] is a member of the network like any other node:
5//! it is tracked on [`Network`](crate::network::Network), written to
6//! `zombie.json`, recreated on attach and destroyed with the namespace. It
7//! has no chain, so the only protocol-level surface it exposes is the set of
8//! ports it declared and where each one is reachable.
9

23/// Where one declared port of a custom process is reachable.
24#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
25pub struct ProcessPort {
26    /// The port the process listens on: as declared, or the one picked for `0`.
27    pub port: Port,
28    /// `host:port` as reachable from where zombienet runs: a port-forward in
29    /// k8s (unless running in CI, where it is the pod's own address), the
30    /// published host port on docker, the port itself on native.
31    pub external: String,
32    /// `host:port` as reachable from inside the network (other nodes and
33    /// processes): the process's Service in k8s (`<name>:<port>`), the
34    /// container ip on docker, the same as `external` on native.
35    pub internal: String,
36}
37
38/// A running custom process.
39#[derive(Clone, Serialize)]
40pub struct CustomProcessNode {
41    #[serde(flatten)]
42    pub(crate) core: NodeCore,
43    pub(crate) spec: CustomProcess,
44    /// Ip the process is reachable at.
45    pub(crate) ip: IpAddr,
46    /// Where each declared port is reachable, by the port's name.
47    pub(crate) ports: BTreeMap<String, ProcessPort>,
48}
49
50/// Deserialization counterpart used when re-attaching to a running network.
51#[derive(Deserialize)]
52pub(crate) struct RawCustomProcessNode {
53    pub(crate) name: String,
54    pub(crate) spec: CustomProcess,
55    pub(crate) ip: IpAddr,
56    #[serde(default)]
57    pub(crate) ports: BTreeMap<String, ProcessPort>,
58    /// The provider's own record of the process, which it needs to reattach.
59    pub(crate) inner: serde_json::Value,
60}
61
62impl CustomProcessNode {
63    pub(crate) fn new(
64        name: impl Into<String>,
65        inner: provider::DynNode,
66        spec: CustomProcess,
67        ip: IpAddr,
68        ports: BTreeMap<String, ProcessPort>,
69    ) -> Self {
70        Self {
71            core: NodeCore::new(name, inner, NodeKind::CustomProcess),
72            spec,
73            ip,
74            ports,
75        }
76    }
77
78    /// The provider-generic part of this process.
79    pub fn core(&self) -> &NodeCore {
80        &self.core
81    }
82
83    pub fn name(&self) -> &str {
84        self.core.name()
85    }
86
87    /// The configuration this process was spawned from.
88    pub fn spec(&self) -> &CustomProcess {
89        &self.spec
90    }
91
92    /// Ip the process is reachable at.
93    pub fn ip(&self) -> IpAddr {
94        self.ip
95    }
96
97    /// Where every declared port is reachable, by the port's name.
98    pub fn ports(&self) -> &BTreeMap<String, ProcessPort> {
99        &self.ports
100    }
101
102    /// Where the port named `name` is reachable, if the process declared it.
103    pub fn port(&self, name: &str) -> Option<&ProcessPort> {
104        self.ports.get(name)
105    }
106
107    /// `host:port` to reach the port named `name` from where zombienet runs,
108    /// if the process declared it.
109    pub fn get_uri_for_name(&self, name: &str) -> Option<&str> {
110        self.ports.get(name).map(|p| p.external.as_str())
111    }
112
113    /// The first declared port, in declaration order, if any.
114    fn first_port(&self) -> Option<&ProcessPort> {
115        self.spec
116            .ports()
117            .first()
118            .and_then(|declared| self.ports.get(&declared.name))
119    }
120
121    /// Check if the process is responsive by connecting to its first declared
122    /// port, with a short timeout (2 seconds). A process that declares no port
123    /// has nothing to probe and is reported responsive.
124    ///
125    /// On kubernetes outside CI the address is a local port-forward, and on
126    /// docker a published port behind docker's proxy; both accept a connection
127    /// whether or not the process listens, so there the probe says the forward
128    /// is up, not that the process answers. Native and kubernetes in CI reach
129    /// the process itself.
130    pub async fn is_responsive(&self) -> bool {
131        match self.first_port() {
132            Some(port) => {
133                tokio::time::timeout(Duration::from_secs(2), wait_tcp_ready(&port.external))
134                    .await
135                    .is_ok()
136            },
137            None => true,
138        }
139    }
140}

142#[async_trait]
143impl SpawnedNode for CustomProcessNode {
144    fn core(&self) -> &NodeCore {
145        &self.core
146    }
147
148    /// A custom process has no readiness definition yet (zombienet-sdk#596):
149    /// a declared port that is slow to open must not fail a network-wide
150    /// wait, so there is nothing to wait on. [`Self::is_responsive`] probes.
151    async fn wait_until_is_up(&self, _timeout_secs: u64) -> Result<(), anyhow::Error> {
152        debug!("[{}] custom process, not waited on", self.name());
153        Ok(())
154    }
155}

157impl std::fmt::Debug for CustomProcessNode {
158    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
159        f.debug_struct("CustomProcessNode")
160            .field("inner", &"inner_skipped")
161            .field("name", &self.name())
162            .field("spec", &self.spec)
163            .field("ip", &self.ip)
164            .field("ports", &self.ports)
165            .finish()
166    }
167}

169#[cfg(test)]
170mod tests {
171    use super::*;
172
173    #[test]
174    fn a_raw_process_without_ports_still_deserializes_but_needs_its_inner() {
175        let json = serde_json::json!({
176            "name": "eth-rpc",
177            "spec": { "name": "eth-rpc", "command": "eth-rpc" },
178            "ip": "127.0.0.1",
179            "inner": { "provider_tag": "k8s" },
180        });
181        let raw: RawCustomProcessNode = serde_json::from_value(json).unwrap();
182        assert_eq!(raw.name, "eth-rpc");
183        assert!(raw.ports.is_empty());
184        assert_eq!(raw.inner["provider_tag"], "k8s");
185
186        let without_inner = serde_json::json!({
187            "name": "eth-rpc",
188            "spec": { "name": "eth-rpc", "command": "eth-rpc" },
189            "ip": "127.0.0.1",
190        });
191        assert!(serde_json::from_value::<RawCustomProcessNode>(without_inner).is_err());
192    }
193
194    #[test]
195    fn ports_round_trip_by_name() {
196        let ports = BTreeMap::from([(
197            "http".to_string(),
198            ProcessPort {
199                port: 8545,
200                external: "127.0.0.1:53421".into(),
201                internal: "10.0.0.7:8545".into(),
202            },
203        )]);
204        let json = serde_json::to_value(&ports).unwrap();
205        assert_eq!(json["http"]["port"], 8545);
206        assert_eq!(json["http"]["external"], "127.0.0.1:53421");
207        let back: BTreeMap<String, ProcessPort> = serde_json::from_value(json).unwrap();
208        assert_eq!(back, ports);
209    }
