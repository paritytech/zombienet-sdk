53    pub async fn from_config(
54        network_config: &NetworkConfig,
55    ) -> Result<NetworkSpec, OrchestratorError> {
56        let mut errs = vec![];
57        let relaychain = network_config
58            .try_relaychain()
59            .map(RelaychainSpec::from_config)
60            .transpose()?;
61        let jamchain = if let Some(jamchain_config) = network_config.jamchain() {
62            Some(JamchainSpec::from_config(jamchain_config)?)
63        } else {
64            None
65        };
66
67        // The chain the parachains are anchored to, either the relaychain or the JAM chain.
68        let root_chain = relaychain
69            .as_ref()
70            .map(|relaychain| relaychain.chain.clone())
71            .or_else(|| jamchain.as_ref().map(|jamchain| jamchain.id.clone()));
72
73        let mut parachains = vec![];
74
75        if let Some(root_chain) = root_chain {
76            // TODO: move to `fold` or map+fold
77            for para_config in network_config.parachains() {
78                match ParachainSpec::from_config(para_config, root_chain.clone()) {
79                    Ok(para) => parachains.push(para),
80                    Err(err) => errs.push(err),
81                }
82            }
83        }
84
85        // Processes and nodes share one registry on the running network. The
86        // names are final here (groups expanded, collisions between nodes
87        // resolved), so this is where a process taking one is caught.
88        let mut names: HashSet<&str> = relaychain
89            .iter()
90            .flat_map(|relay| relay.nodes.iter().map(|node| node.name.as_str()))
91            .chain(
92                jamchain
93                    .iter()
94                    .flat_map(|jam| jam.nodes.iter().map(|node| node.name.as_str())),
95            )
96            .chain(
97                parachains
98                    .iter()
99                    .flat_map(|para| para.collators.iter().map(|node| node.name.as_str())),
100            )
101            .collect();
102        for process in network_config.custom_processes() {
103            if !names.insert(process.name()) {
104                errs.push(OrchestratorError::InvalidConfig(format!(
105                    "custom_processes['{}'].name: already used by a node or custom process",
106                    process.name()
107                )));
108            }
109        }
110
111        if errs.is_empty() {
112            Ok(NetworkSpec {
113                relaychain,
114                jamchain,
115                parachains,
116                hrmp_channels: network_config
117                    .hrmp_channels()
118                    .into_iter()
119                    .cloned()
120                    .collect(),
121                global_settings: network_config.global_settings().clone(),
122                custom_processes: network_config
123                    .custom_processes()
124                    .into_iter()
125                    .cloned()
126                    .collect(),
127            })
128        } else {
129            Err(merge_errs(&errs))
130        }
131    }

133    pub async fn populate_nodes_available_args(
134        &mut self,
135        ns: Arc<dyn ProviderNamespace + Send + Sync>,
136    ) -> Result<(), OrchestratorError> {
137        let network_nodes = self.collect_network_nodes();
138
139        let mut image_command_to_nodes_mapping =
140            Self::create_image_command_to_nodes_mapping(network_nodes);
141
142        let available_args_outputs =
143            Self::retrieve_all_nodes_available_args_output(ns, &image_command_to_nodes_mapping)
144                .await?;
145
146        Self::update_nodes_available_args_output(
147            &mut image_command_to_nodes_mapping,
148            available_args_outputs,
149        );
150
151        Ok(())
152    }

155    pub async fn node_available_args_output(
156        &self,
157        node_spec: &NodeSpec,
158        ns: Arc<dyn ProviderNamespace + Send + Sync>,
159    ) -> Result<String, ProviderError> {
160        // try to find a node that use the same combination of image/cmd
161        let cmp_fn = |ad_hoc: &&NodeSpec| -> bool {
162            ad_hoc.image == node_spec.image && ad_hoc.command == node_spec.command
163        };
164
165        // check if we already had computed the args output for this cmd/[image]
166        let node = self
167            .relaychain
168            .iter()
169            .flat_map(|relaychain| relaychain.nodes.iter())
170            .find(cmp_fn);
171        let node = if let Some(node) = node {
172            Some(node)
173        } else {
174            let node = self
175                .parachains
176                .iter()
177                .find_map(|para| para.collators.iter().find(cmp_fn));
178
179            node
180        };
181
182        let output = if let Some(node) = node {
183            node.available_args_output.clone().expect(&format!(
184                "args_output should be set for running nodes {THIS_IS_A_BUG}"
185            ))
186        } else {
187            // we need to compute the args output
188            let image = node_spec
189                .image
190                .as_ref()
191                .map(|image| image.as_str().to_string());
192            let command = node_spec.command.as_str().to_string();
193
194            ns.get_node_available_args((command, image)).await?
195        };
196
197        Ok(output)
198    }

229    pub async fn build_parachain_artifacts<'a, T: FileSystem>(
230        &mut self,
231        ns: DynNamespace,
232        scoped_fs: &ScopedFilesystem<'a, T>,
233        relaychain_id: &str,
234        base_dir_exists: bool,
235    ) -> Result<(), anyhow::Error> {
236        for para in self.parachains.iter_mut() {
237            let chain_spec_raw_path = para
238                .build_chain_spec(
239                    relaychain_id,
240                    &ns,
241                    scoped_fs,
242                    para.post_process_script.clone().as_deref(),
243                )
244                .await?;
245
246            trace!("creating dirs for {}", &para.unique_id);
247            if base_dir_exists {
248                scoped_fs.create_dir_all(&para.unique_id).await?;
249            } else {
250                scoped_fs.create_dir(&para.unique_id).await?;
251            };
252            trace!("created dirs for {}", &para.unique_id);
253
254            // create wasm/state
255            para.genesis_state
256                .build(
257                    chain_spec_raw_path.clone(),
258                    format!("{}/genesis-state", para.unique_id),
259                    &ns,
260                    scoped_fs,
261                    None,
262                )
263                .await?;
264            debug!("parachain genesis state built!");
265            para.genesis_wasm
266                .build(
267                    chain_spec_raw_path,
268                    format!("{}/genesis-wasm", para.unique_id),
269                    &ns,
270                    scoped_fs,
271                    None,
272                )
273                .await?;
274            debug!("parachain genesis wasm built!");
275        }
276
277        Ok(())
278    }

280    // collect mutable references to all nodes from relaychain and parachains
281    fn collect_network_nodes(&mut self) -> Vec<&mut NodeSpec> {
282        vec![
283            self.relaychain
284                .iter_mut()
285                .flat_map(|relaychain| relaychain.nodes.iter_mut())
286                .collect::<Vec<_>>(),
287            self.parachains
288                .iter_mut()
289                .flat_map(|para| para.collators.iter_mut())
290                .collect(),
291        ]
292        .into_iter()
293        .flatten()
294        .collect::<Vec<_>>()
295    }

297    // initialize the mapping of all possible node image/commands to corresponding nodes
298    fn create_image_command_to_nodes_mapping(
299        network_nodes: Vec<&mut NodeSpec>,
300    ) -> HashMap<(Option<String>, String), Vec<&mut NodeSpec>> {
301        network_nodes.into_iter().fold(
302            HashMap::new(),
303            |mut acc: HashMap<(Option<String>, String), Vec<&mut node::NodeSpec>>, node| {
304                // build mapping key using image and command if image is present or command only
305                let key = node
306                    .image
307                    .as_ref()
308                    .map(|image| {
309                        (
310                            Some(image.as_str().to_string()),
311                            node.command.as_str().to_string(),
312                        )
313                    })
314                    .unwrap_or_else(|| (None, node.command.as_str().to_string()));
315
316                // append the node to the vector of nodes for this image/command tuple
317                if let Entry::Vacant(entry) = acc.entry(key.clone()) {
318                    entry.insert(vec![node]);
319                } else {
320                    acc.get_mut(&key).unwrap().push(node);
321                }
322
323                acc
324            },
325        )
326    }

328    async fn retrieve_all_nodes_available_args_output(
329        ns: Arc<dyn ProviderNamespace + Send + Sync>,
330        image_command_to_nodes_mapping: &HashMap<(Option<String>, String), Vec<&mut NodeSpec>>,
331    ) -> Result<Vec<(Option<String>, String, String)>, OrchestratorError> {
332        try_join_all(
333            image_command_to_nodes_mapping
334                .keys()
335                .map(|(image, command)| async {
336                    let image = image.clone();
337                    let command = command.clone();
338                    // get node available args output from image/command
339                    let available_args = ns
340                        .get_node_available_args((command.clone(), image.clone()))
341                        .await?;
342                    debug!(
343                        "retrieved available args for image: {:?}, command: {}",
344                        image, command
345                    );
346
347                    // map the result to include image and command
348                    Ok::<_, OrchestratorError>((image, command, available_args))
349                })
350                .collect::<Vec<_>>(),
351        )
352        .await
353    }
354
355    fn update_nodes_available_args_output(
356        image_command_to_nodes_mapping: &mut HashMap<(Option<String>, String), Vec<&mut NodeSpec>>,
357        available_args_outputs: Vec<(Option<String>, String, String)>,
358    ) {
359        for (image, command, available_args_output) in available_args_outputs {
360            let nodes = image_command_to_nodes_mapping
361                .get_mut(&(image, command))
362                .expect(&format!(
363                    "node image/command key should exist {THIS_IS_A_BUG}"
364                ));
365
366            for node in nodes {
367                node.available_args_output = Some(available_args_output.clone());
368            }
369        }
370    }
