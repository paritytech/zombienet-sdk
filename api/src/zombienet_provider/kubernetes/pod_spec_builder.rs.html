20    #[allow(clippy::too_many_arguments)]
21    pub(super) fn build(
22        name: &str,
23        image: &str,
24        resources: Option<&Resources>,
25        program: &str,
26        args: &[String],
27        env: &[(String, String)],
28        ports: &[(String, Port)],
29        readiness: Option<&ReadinessProbe>,
30        wrapper: bool,
31    ) -> PodSpec {
32        let tolerations = if let Ok(node_type) = env::var("X_INFRA_NODETYPE") {
33            let t = Toleration {
34                effect: Some("NoExecute".into()),
35                key: Some("nodetype".into()),
36                operator: Some("Equal".into()),
37                value: Some(node_type),
38                ..Default::default()
39            };
40            Some(vec![t])
41        } else {
42            None
43        };
44
45        PodSpec {
46            hostname: Some(name.to_string()),
47            init_containers: Some(vec![Self::build_helper_binaries_setup_container()]),
48            containers: vec![Self::build_main_container(
49                name, image, resources, program, args, env, ports, readiness, wrapper,
50            )],
51            volumes: Some(Self::build_volumes()),
52            tolerations,
53            // A direct process that exits is done, by success or by failure: the
54            // pod ends with it, and so says what happened, instead of the
55            // kubelet restarting it in a loop. A wrapped node keeps its pod up
56            // through the wrapper either way.
57            restart_policy: (!wrapper).then(|| "Never".to_string()),
58            ..Default::default()
59        }
60    }
61
62    #[allow(clippy::too_many_arguments)]
63    fn build_main_container(
64        name: &str,
65        image: &str,
66        resources: Option<&Resources>,
67        program: &str,
68        args: &[String],
69        env: &[(String, String)],
70        ports: &[(String, Port)],
71        readiness: Option<&ReadinessProbe>,
72        wrapper: bool,
73    ) -> Container {
74        // With the wrapper, `program` is started on demand through the pipe the
75        // wrapper listens on. Without it the container runs `program` directly,
76        // as the image would, so an off-the-shelf image needs no `bash`.
77        let command = if wrapper {
78            [
79                vec!["/zombie-wrapper.sh".to_string(), program.to_string()],
80                args.to_vec(),
81            ]
82            .concat()
83        } else {
84            [vec![program.to_string()], args.to_vec()].concat()
85        };
86
87        let container_ports = if ports.is_empty() {
88            None
89        } else {
90            Some(
91                ports
92                    .iter()
93                    .map(|(name, port)| ContainerPort {
94                        name: Some(name.clone()),
95                        container_port: i32::from(*port),
96                        ..Default::default()
97                    })
98                    .collect(),
99            )
100        };
101
102        // A wrapped node reports readiness itself (the wrapper is up at once
103        // and the node is waited on through its metrics). A direct process is
104        // Ready, and so routed to by its Service, once the declared probe
105        // passes on the pod address.
106        let readiness_probe = match (wrapper, readiness) {
107            (false, Some(probe)) => {
108                let (tcp_socket, http_get) = match &probe.target {
109                    ReadinessTarget::Tcp(port) => (
110                        Some(TCPSocketAction {
111                            port: IntOrString::Int(i32::from(*port)),
112                            ..Default::default()
113                        }),
114                        None,
115                    ),
116                    ReadinessTarget::Http { port, path } => (
117                        None,
118                        Some(HTTPGetAction {
119                            port: IntOrString::Int(i32::from(*port)),
120                            path: Some(path.clone()),
121                            ..Default::default()
122                        }),
123                    ),
124                };
125                Some(Probe {
126                    tcp_socket,
127                    http_get,
128                    period_seconds: Some(i32::try_from(probe.period_secs).unwrap_or(i32::MAX)),
129                    // the same patience as the probe zombienet runs itself
130                    timeout_seconds: Some(5),
131                    failure_threshold: Some(3),
132                    ..Default::default()
133                })
134            },
135            _ => None,
136        };
137
138        let extra_mounts = if wrapper {
139            vec![VolumeMount {
140                name: "zombie-wrapper-volume".to_string(),
141                mount_path: "/zombie-wrapper.sh".to_string(),
142                sub_path: Some("zombie-wrapper.sh".to_string()),
143                ..Default::default()
144            }]
145        } else {
146            vec![]
147        };
148
149        Container {
150            name: name.to_string(),
151            image: Some(image.to_string()),
152            image_pull_policy: Some("Always".to_string()),
153            command: Some(command),
154            ports: container_ports,
155            readiness_probe,
156            env: Some(
157                env.iter()
158                    .map(|(name, value)| EnvVar {
159                        name: name.clone(),
160                        value: Some(value.clone()),
161                        value_from: None,
162                    })
163                    .collect(),
164            ),
165            volume_mounts: Some(Self::build_volume_mounts(extra_mounts)),
166            resources: Self::build_resources_requirements(resources),
167            ..Default::default()
168        }
169    }

262    fn build_resources_requirements(resources: Option<&Resources>) -> Option<ResourceRequirements> {
263        resources.map(|resources| ResourceRequirements {
264            limits: Self::build_resources_requirements_quantities(
265                resources.limit_cpu(),
266                resources.limit_memory(),
267            ),
268            requests: Self::build_resources_requirements_quantities(
269                resources.request_cpu(),
270                resources.request_memory(),
271            ),
272            ..Default::default()
273        })
274    }
275
276    fn build_resources_requirements_quantities(
277        cpu: Option<&ResourceQuantity>,
278        memory: Option<&ResourceQuantity>,
279    ) -> Option<BTreeMap<String, Quantity>> {
280        let mut quantities = BTreeMap::new();
281
282        if let Some(cpu) = cpu {
283            quantities.insert("cpu".to_string(), Quantity(cpu.as_str().to_string()));
284        }
285
286        if let Some(memory) = memory {
287            quantities.insert("memory".to_string(), Quantity(memory.as_str().to_string()));
288        }
289
290        if !quantities.is_empty() {
291            Some(quantities)
292        } else {
293            None
294        }
295    }

302    fn main_container(spec: &PodSpec) -> &Container {
303        spec.containers

309    #[test]
310    fn a_wrapped_node_runs_through_the_wrapper_and_has_no_probe() {
311        let spec = PodSpecBuilder::build(
312            "proc",
313            "img",
314            None,
315            "polkadot",
316            &["--dev".to_string()],
317            &[],
318            &[],
319            None,
320            true,
321        );
322        assert!(spec.restart_policy.is_none());
323        let container = main_container(&spec);
324        assert_eq!(
325            container.command.as_deref().unwrap(),
326            ["/zombie-wrapper.sh", "polkadot", "--dev"]
327        );
328        assert!(container.ports.is_none());
329        assert!(container.readiness_probe.is_none());
330        assert!(container
331            .volume_mounts
332            .as_ref()
333            .unwrap()
334            .iter()
335            .any(|m| m.name == "zombie-wrapper-volume"));
336    }

338    #[test]
339    fn a_direct_process_runs_its_command_exposes_ports_and_probes_as_told() {
340        let ports = vec![("api".to_string(), 5001), ("gateway".to_string(), 8080)];
341        let spec = PodSpecBuilder::build(
342            "proc",
343            "img",
344            None,
345            "ipfs",
346            &["daemon".to_string()],
347            &[],
348            &ports,
349            Some(&ReadinessProbe {
350                target: ReadinessTarget::Http {
351                    port: 8080,
352                    path: "/version".into(),
353                },
354                period_secs: 3,
355            }),
356            false,
357        );
358        assert_eq!(spec.restart_policy.as_deref(), Some("Never"));
359        let container = main_container(&spec);
360        assert_eq!(container.command.as_deref().unwrap(), ["ipfs", "daemon"]);
361
362        let container_ports = container.ports.as_ref().unwrap();
363        assert_eq!(
364            container_ports
365                .iter()
366                .map(|p| (p.name.as_deref().unwrap(), p.container_port))
367                .collect::<Vec<_>>(),
368            [("api", 5001), ("gateway", 8080)]
369        );
370
371        let probe = container.readiness_probe.as_ref().unwrap();
372        let http = probe.http_get.as_ref().unwrap();
373        assert_eq!(http.port, IntOrString::Int(8080));
374        assert_eq!(http.path.as_deref(), Some("/version"));
375        assert_eq!(probe.period_seconds, Some(3));
376        assert_eq!(probe.timeout_seconds, Some(5));
377        assert!(!container
378            .volume_mounts
379            .as_ref()
380            .unwrap()
381            .iter()
382            .any(|m| m.name == "zombie-wrapper-volume"));
383    }
