17pub fn generate(port: Option<Port>) -> Result<ParkedPort, GeneratorError> {
18    let port = port.unwrap_or(0);
19    let addr: SocketAddr = format!("[::]:{port}")
20        .parse()
21        .expect("addr should be valid");
22    let socket = Socket::new(Domain::IPV6, Type::STREAM, Some(Protocol::TCP))
23        .map_err(|_e| GeneratorError::PortGeneration(port, "Can't create the socket".into()))?;
24
25    // Explicitly disable v6only for dual-stack support
26    socket.set_only_v6(false).map_err(|_e| {
27        GeneratorError::PortGeneration(port, "Can't set v6 only to false in socket".into())
28    })?;
29    socket
30        .bind(&addr.into())
31        .map_err(|_e| GeneratorError::PortGeneration(port, "Can't bind in socket".into()))?;
32    socket
33        .listen(128)
34        .map_err(|_e| GeneratorError::PortGeneration(port, "Can't listen in socket".into()))?;
35
36    let listener: TcpListener = socket.into();
37    let port = listener
38        .local_addr()
39        .expect(&format!(
40            "We should always get the local_addr from the listener {THIS_IS_A_BUG}"
41        ))
42        .port();
43    Ok(ParkedPort::new(port, listener))
44}
