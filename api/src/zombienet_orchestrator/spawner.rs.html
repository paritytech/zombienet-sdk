28#[derive(Clone)]
29pub struct SpawnNodeCtx<'a, T: FileSystem> {
30    /// Relaychain id, from the chain-spec (e.g rococo_local_testnet)
31    pub(crate) chain_id: &'a str,
32    // Parachain id, from the chain-spec (e.g local_testnet)
33    pub(crate) parachain_id: Option<&'a str>,
34    /// Relaychain chain name (e.g rococo-local)
35    pub(crate) chain: &'a str,
36    /// Role of the node in the network
37    pub(crate) role: ZombieRole,
38    /// Ref to the namespace
39    pub(crate) ns: &'a DynNamespace,
40    /// Ref to an scoped filesystem (encapsulate fs actions inside the ns directory)
41    pub(crate) scoped_fs: &'a ScopedFilesystem<'a, T>,
42    /// Ref to a parachain (used to spawn collators)
43    pub(crate) parachain: Option<&'a ParachainSpec>,
44    /// The string representation of the bootnode address to pass to nodes
45    pub(crate) bootnodes_addr: &'a Vec<String>,
46    /// Flag to wait node is ready or not
47    /// Ready state means we can query Prometheus internal server
48    pub(crate) wait_ready: bool,
49    /// A json representation of the running nodes with their names as 'key'
50    pub(crate) nodes_by_name: serde_json::Value,
51    /// A ref to the global settings
52    pub(crate) global_settings: &'a GlobalSettings,
53    /// `db_snapshot` `AssetLocation`s resolved to local cache paths,
54    /// populated once before the parallel spawn fanout. The provider's
55    /// `initialize_db_snapshot` reads from these paths only — no
56    /// downloads happen inside the per-node spawn.
57    pub(crate) resolved_db_snapshots: &'a ResolvedDbSnapshots,
58}

60pub async fn spawn_node<'a, T>(
61    node: &NodeSpec,
62    mut files_to_inject: Vec<TransferedFile>,
63    ctx: &SpawnNodeCtx<'a, T>,
64) -> Result<NetworkNode, anyhow::Error>
65where
66    T: FileSystem,
67{
68    let mut created_paths = vec![];
69    // Create and inject the keystore IFF
70    // - The node is validator in the relaychain
71    // - The node is collator (encoded as validator) and the parachain is cumulus_based
72    // (parachain_id) should be set then.
73    if node.is_validator && (ctx.parachain.is_none() || ctx.parachain_id.is_some()) {
74        // Generate keystore for node
75        let node_files_path = if let Some(para) = ctx.parachain {
76            para.id.to_string()
77        } else {
78            node.name.clone()
79        };
80        let asset_hub_polkadot = ctx
81            .parachain_id
82            .map(|id| id.starts_with("asset-hub-polkadot"))
83            .unwrap_or_default();
84        let keystore_key_types = node.keystore_key_types.iter().map(String::as_str).collect();
85        let key_filenames = generators::generate_node_keystore(
86            &node.accounts,
87            &node_files_path,
88            ctx.scoped_fs,
89            asset_hub_polkadot,
90            keystore_key_types,
91        )
92        .await
93        .unwrap();
94
95        // Paths returned are relative to the base dir, we need to convert into
96        // fullpaths to inject them in the nodes.
97        let remote_keystore_chain_id = if let Some(id) = ctx.parachain_id {
98            id
99        } else {
100            ctx.chain_id
101        };
102
103        let keystore_path = node.keystore_path.clone().unwrap_or(PathBuf::from(format!(
104            "/data/chains/{remote_keystore_chain_id}/keystore",
105        )));
106
107        for key_filename in key_filenames {
108            let f = TransferedFile::new(
109                PathBuf::from(format!(
110                    "{}/{}/{}",
111                    ctx.ns.base_dir().to_string_lossy(),
112                    node_files_path,
113                    key_filename.to_string_lossy()
114                )),
115                keystore_path.join(key_filename),
116            );
117            files_to_inject.push(f);
118        }
119        created_paths.push(keystore_path);
120    }
121
122    let base_dir = format!("{}/{}", ctx.ns.base_dir().to_string_lossy(), node.name);
123
124    let (cfg_path, data_path, relay_data_path) = if !ctx.ns.capabilities().prefix_with_full_path {
125        (
126            NODE_CONFIG_DIR.into(),
127            NODE_DATA_DIR.into(),
128            NODE_RELAY_DATA_DIR.into(),
129        )
130    } else {
131        let cfg_path = format!("{}{NODE_CONFIG_DIR}", base_dir);
132        let data_path = format!("{}{NODE_DATA_DIR}", base_dir);
133        let relay_data_path = format!("{}{NODE_RELAY_DATA_DIR}", base_dir);
134        (cfg_path, data_path, relay_data_path)
135    };
136
137    let gen_opts = generators::GenCmdOptions {
138        relay_chain_name: ctx.chain.to_string(),
139        cfg_path,           // TODO: get from provider/ns
140        data_path,          // TODO: get from provider
141        relay_data_path,    // TODO: get from provider
142        use_wrapper: false, // TODO: get from provider
143        bootnode_addr: ctx.bootnodes_addr.clone(),
144        use_default_ports_in_cmd: ctx.ns.capabilities().use_default_ports_in_cmd,
145        // IFF the provider require an image (e.g k8s) we know this is not native
146        is_native: !ctx.ns.capabilities().requires_image,
147    };
148
149    let mut collator_full_node_prom_port: Option<u16> = None;
150    let mut collator_full_node_prom_port_external: Option<u16> = None;
151
152    let (program, args) = match ctx.role {
153        // Collator should be `non-cumulus` one (e.g adder/undying)
154        ZombieRole::Node | ZombieRole::Collator => {
155            let maybe_para_id = ctx.parachain.map(|para| para.id);
156
157            generators::generate_node_command(node, gen_opts.clone(), maybe_para_id)
158        },
159        ZombieRole::CumulusCollator => {
160            let para = ctx.parachain.expect(&format!(
161                "parachain must be part of the context {THIS_IS_A_BUG}"
162            ));
163            collator_full_node_prom_port = node.full_node_prometheus_port.as_ref().map(|p| p.0);
164
165            generators::generate_node_command_cumulus(node, gen_opts.clone(), para.id)
166        },
167        _ => unreachable!(), /* TODO: do we need those?
168                              * ZombieRole::Bootnode => todo!(),
169                              * ZombieRole::Companion => todo!(), */
170    };
171
172    // apply running networ replacements
173    let args: Vec<String> = args
174        .iter()
175        .map(|arg| apply_running_network_replacements(arg, &ctx.nodes_by_name))
176        .collect();
177
178    info!(
179        "🚀 {}, spawning.... with command: {} {}",
180        node.name,
181        program,
182        args.join(" ")
183    );
184
185    let ports = if ctx.ns.capabilities().use_default_ports_in_cmd {
186        // should use default ports to as internal
187        [
188            (P2P_PORT, node.p2p_port.0),
189            (RPC_PORT, node.rpc_port.0),
190            (PROMETHEUS_PORT, node.prometheus_port.0),
191        ]
192    } else {
193        [
194            (P2P_PORT, P2P_PORT),
195            (RPC_PORT, RPC_PORT),
196            (PROMETHEUS_PORT, PROMETHEUS_PORT),
197        ]
198    };
199
200    let resolved_db_snapshot = node
201        .db_snapshot
202        .as_ref()
203        .and_then(|loc| ctx.resolved_db_snapshots.get(loc).cloned());
204    if node.db_snapshot.is_some() && resolved_db_snapshot.is_none() {
205        // Invariant: every NodeSpec.db_snapshot must have been resolved
206        // by orchestrator::generators::resolve_db_snapshots before we
207        // get here. Hitting this path means the resolution step was
208        // skipped for this node.
209        return Err(anyhow::anyhow!(
210            "{THIS_IS_A_BUG}: node {} has db_snapshot {:?} but no resolved cache entry",
211            node.name,
212            node.db_snapshot,
213        ));
214    }
215
216    let spawn_ops = SpawnNodeOptions::new(node.name.clone(), program)
217        .args(args)
218        .env(
219            node.env
220                .iter()
221                .map(|var| (var.name.clone(), var.value.clone())),
222        )
223        .injected_files(files_to_inject)
224        .created_paths(created_paths)
225        .db_snapshot(resolved_db_snapshot)
226        .port_mapping(HashMap::from(ports))
227        .node_log_path(node.node_log_path.clone());
228
229    let spawn_ops = if let Some(image) = node.image.as_ref() {
230        spawn_ops.image(image.as_str())
231    } else {
232        spawn_ops
233    };
234
235    let spawn_ops = if let Some(resources) = node.resources.as_ref() {
236        spawn_ops.resources(resources.clone())
237    } else {
238        spawn_ops
239    };
240
241    // Drops the port parking listeners before spawn
242    node.ws_port.drop_listener();
243    node.p2p_port.drop_listener();
244    node.rpc_port.drop_listener();
245    node.prometheus_port.drop_listener();
246    if let Some(port) = &node.full_node_p2p_port {
247        port.drop_listener();
248    }
249    if let Some(port) = &node.full_node_prometheus_port {
250        port.drop_listener();
251    }
252
253    let running_node = ctx.ns.spawn_node(&spawn_ops).await.with_context(|| {
254        format!(
255            "Failed to spawn node: {} with opts: {:#?}",
256            node.name, spawn_ops
257        )
258    })?;
259
260    let mut ip_to_use = if let Some(local_ip) = ctx.global_settings.local_ip() {
261        *local_ip
262    } else {
263        LOCALHOST
264    };
265
266    let (rpc_port_external, prometheus_port_external, p2p_external);
267
268    if running_in_ci() && ctx.ns.provider_name() == "k8s" {
269        // running kubernets in ci require to use ip and default port
270        (rpc_port_external, prometheus_port_external, p2p_external) =
271            (RPC_PORT, PROMETHEUS_PORT, P2P_PORT);
272        collator_full_node_prom_port_external = Some(FULL_NODE_PROMETHEUS_PORT);
273        ip_to_use = running_node.ip().await?;
274    } else {
275        // Create port-forward iff we are not in CI or provider doesn't use the default ports (native)
276        let ports = futures::future::try_join_all(vec![
277            running_node.create_port_forward(node.rpc_port.0, RPC_PORT),
278            running_node.create_port_forward(node.prometheus_port.0, PROMETHEUS_PORT),
279        ])
280        .await?;
281
282        (rpc_port_external, prometheus_port_external, p2p_external) = (
283            ports[0].unwrap_or(node.rpc_port.0),
284            ports[1].unwrap_or(node.prometheus_port.0),
285            // p2p don't need port-fwd
286            node.p2p_port.0,
287        );
288
289        if let Some(full_node_prom_port) = collator_full_node_prom_port {
290            let port_fwd = running_node
291                .create_port_forward(full_node_prom_port, FULL_NODE_PROMETHEUS_PORT)
292                .await?;
293            collator_full_node_prom_port_external = Some(port_fwd.unwrap_or(full_node_prom_port));
294        }
295    }
296
297    let multiaddr = generators::generate_node_bootnode_addr(
298        &node.peer_id,
299        &running_node.ip().await?,
300        if ctx.ns.provider_name() == "k8s" {
301            P2P_PORT
302        } else {
303            p2p_external
304        }, // for k8s use always the internal port
305        running_node.args().as_ref(),
306        &node.p2p_cert_hash,
307    )?;
308
309    let ws_uri = format!("ws://{ip_to_use}:{rpc_port_external}");
310    let prometheus_uri = format!("http://{ip_to_use}:{prometheus_port_external}/metrics");
311    info!("🚀 {}, should be running now", node.name);
312    info!(
313        "💻 {}: direct link (pjs) https://polkadot.js.org/apps/?rpc={ws_uri}#/explorer",
314        node.name
315    );
316    info!(
317        "💻 {}: direct link (papi) https://dev.papi.how/explorer#networkId=custom&endpoint={ws_uri}",
318        node.name
319    );
320
321    info!("📊 {}: metrics link {prometheus_uri}", node.name);
322
323    if let Some(full_node_prom_port) = collator_full_node_prom_port_external {
324        info!(
325            "📊 {}: collator full-node metrics link http://{}:{}/metrics",
326            node.name, ip_to_use, full_node_prom_port
327        );
328    }
329
330    info!("📓 logs cmd: {}", running_node.log_cmd());
331
332    let node_ctx = if let Some(parachain) = ctx.parachain {
333        NodeContext::Para {
334            para_id: parachain.id,
335            is_cumulus_based: parachain.is_cumulus_based,
336        }
337    } else {
338        NodeContext::Rc
339    };
340
341    Ok(NetworkNode::new(
342        node.name.clone(),
343        ws_uri,
344        prometheus_uri,
345        multiaddr,
346        node.clone(),
347        running_node,
348        gen_opts,
349        node_ctx,
350    ))
351}

353pub async fn spawn_process(
354    custom_process: &CustomProcess,
355    ns: Arc<dyn ProviderNamespace + Send + Sync>,
356) -> Result<(), anyhow::Error> {
357    let args: Vec<String> = custom_process
358        .args()
359        .iter()
360        .flat_map(|arg| arg.to_vec())
361        .collect();
362
363    let spawn_ops = SpawnNodeOptions::new(custom_process.name(), custom_process.command().as_str())
364        .args(&args)
365        .env(
366            custom_process
367                .env()
368                .iter()
369                .map(|var| (var.name.clone(), var.value.clone())),
370        );
371
372    let spawn_ops = if let Some(image) = custom_process.image() {
373        spawn_ops.image(image.as_str())
374    } else {
375        spawn_ops
376    };
377
378    info!(
379        "🚀 {}, spawning custom process.... with command: {} {}",
380        custom_process.name(),
381        custom_process.command().as_str(),
382        args.join(" ")
383    );
384
385    let running_node = ns.spawn_node(&spawn_ops).await.with_context(|| {
386        format!(
387            "Failed to spawn node: {} with opts: {:#?}",
388            custom_process.name(),
389            spawn_ops
390        )
391    })?;
392
393    info!("📓 logs cmd: {}", running_node.log_cmd());
394
395    Ok(())
396}

398pub async fn spawn_jam_node<'a, T>(
399    node: &JamNodeSpec,
400    mut files_to_inject: Vec<TransferedFile>,
401    ctx: &SpawnNodeCtx<'a, T>,
402) -> Result<JamNetworkNode, anyhow::Error>
403where
404    T: FileSystem,
405{
406    let base_dir = format!("{}/{}", ctx.ns.base_dir().to_string_lossy(), node.name);
407
408    let (cfg_path, data_path) = if !ctx.ns.capabilities().prefix_with_full_path {
409        (NODE_CONFIG_DIR.into(), NODE_DATA_DIR.into())
410    } else {
411        let cfg_path = format!("{}{NODE_CONFIG_DIR}", base_dir);
412        let data_path = format!("{}{NODE_DATA_DIR}", base_dir);
413        (cfg_path, data_path)
414    };
415
416    // create local seed file and set to transfer process:
417
418    // 1. Create local dir to store all the needed files
419    ctx.scoped_fs
420        .create_dir_all(PathBuf::from(&node.name))
421        .await?;
422
423    // 2. set paths to use (local and remote)
424    let keys_remote = PathBuf::from(format!("{}/{}/keys", NODE_CONFIG_DIR, ctx.chain));
425    let seed_local_path = PathBuf::from(format!("{}/{}.seed", base_dir, node.name));
426    let seed_remote_path = PathBuf::from(format!(
427        "{}/{}.seed",
428        keys_remote.to_string_lossy(),
429        node.name
430    ));
431
432    // 3. write local seed
433    ctx.scoped_fs
434        .write(&seed_local_path, hash_raw(node.accounts.seed.as_bytes()))
435        .await?;
436
437    // 4. set paths/files to use in remote.
438    let created_paths = vec![keys_remote];
439    files_to_inject.push(TransferedFile::new(seed_local_path, seed_remote_path));
440
441    let gen_opts = generators::GenCmdOptions {
442        relay_chain_name: ctx.chain.to_string(),
443        cfg_path,  // TODO: get from provider/ns
444        data_path, // TODO: get from provider
445        use_default_ports_in_cmd: ctx.ns.capabilities().use_default_ports_in_cmd,
446        // IFF the provider require an image (e.g k8s) we know this is not native
447        is_native: !ctx.ns.capabilities().requires_image,
448        bootnode_addr: ctx.bootnodes_addr.clone(),
449        ..Default::default()
450    };
451
452    let (program, args) = generators::generate_jam_node_command(node, gen_opts.clone());
453    // apply running networ replacements
454    let args: Vec<String> = args
455        .iter()
456        .map(|arg| apply_running_network_replacements(arg, &ctx.nodes_by_name))
457        .collect();
458
459    info!(
460        "🚀 {}, spawning.... with command: {} {}",
461        node.name,
462        program,
463        args.join(" ")
464    );
465
466    let ports = if ctx.ns.capabilities().use_default_ports_in_cmd {
467        // should use default ports to as internal
468        [(JAM_PORT, node.port.0), (RPC_PORT, node.rpc_port.0)]
469    } else {
470        [(JAM_PORT, JAM_PORT), (RPC_PORT, RPC_PORT)]
471    };
472
473    let spawn_ops = SpawnNodeOptions::new(node.name.clone(), program)
474        .args(args)
475        .env(
476            node.env
477                .iter()
478                .map(|var| (var.name.clone(), var.value.clone())),
479        )
480        .injected_files(files_to_inject)
481        .created_paths(created_paths)
482        .port_mapping(HashMap::from(ports));
483
484    let spawn_ops = if let Some(image) = node.image.as_ref() {
485        spawn_ops.image(image.as_str())
486    } else {
487        spawn_ops
488    };
489
490    let spawn_ops = if let Some(resources) = node.resources.as_ref() {
491        spawn_ops.resources(resources.clone())
492    } else {
493        spawn_ops
494    };
495
496    // Drops the port parking listeners before spawn
497    node.port.drop_listener();
498    node.rpc_port.drop_listener();
499
500    let running_node = ctx.ns.spawn_node(&spawn_ops).await.with_context(|| {
501        format!(
502            "Failed to spawn node: {} with opts: {:#?}",
503            node.name, spawn_ops
504        )
505    })?;
506
507    let ip_to_use = if let Some(local_ip) = ctx.global_settings.local_ip() {
508        *local_ip
509    } else {
510        LOCALHOST
511    };
512
513    // NOTE: running in ci (k8s) is not supported yet for JAM, so we don't
514    // create port-forwards nor use the internal ip/default ports here.
515
516    info!("🚀 {}, should be running now", node.name);
517    match node.mode {
518        JamNodeMode::Ordinary => {
519            info!("💻 {}, rpc  {ip_to_use}:{}", node.name, node.rpc_port.0);
520        },
521        JamNodeMode::Validator | JamNodeMode::Proxy => {
522            info!(
523                "💻 {}, peer details {}@{ip_to_use}:{}",
524                node.name, node.peer_id, node.port.0
525            );
526        },
527    }
528
529    info!("📓 logs cmd: {}", running_node.log_cmd());
530
531    Ok(JamNetworkNode::new(
532        node.name.clone(),
533        running_node,
534        node.clone(),
535        ip_to_use,
536        gen_opts,
537    ))
