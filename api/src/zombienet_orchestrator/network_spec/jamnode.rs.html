19/// A node configuration, with fine-grained configuration options.
20#[derive(Debug, Clone, Default, Serialize, Deserialize)]
21pub struct JamNodeSpec {
22    // Node name (should be unique or an index will be appended).
23    pub(crate) name: String,
24
25    /// node mode.
26    pub(crate) mode: JamNodeMode,
27
28    // libp2p local identity
29    pub(crate) peer_id: String,
30
31    /// Accounts to be injected in the keystore.
32    pub(crate) accounts: NodeAccounts,
33
34    /// Image to run (only podman/k8s). Override the default.
35    pub(crate) image: Option<Image>,
36
37    /// Command to run the node. Override the default.
38    pub(crate) command: Command,
39
40    /// Optional subcommand for the node.
41    pub(crate) subcommand: Option<Command>,
42
43    /// Arguments to use for node. Appended to default.
44    pub(crate) args: Vec<Arg>,
45
46    /// Environment variables to set (inside pod for podman/k8s, inside shell for native).
47    pub(crate) env: Vec<EnvVar>,
48
49    /// Default resources. Override the default.
50    pub(crate) resources: Option<Resources>,
51
52    /// Port to use.
53    pub(crate) port: ParkedPort,
54
55    /// RPC port to use.
56    pub(crate) rpc_port: ParkedPort,
57
58    /// Telemetry endpoint
59    pub(crate) telemetry_endpoint: Option<String>,
60}

62impl JamNodeSpec {
63    pub fn from_config(
64        node_config: &JamNodeConfig,
65        chain_context: &ChainDefaultContext,
66    ) -> Result<Self, OrchestratorError> {
67        // Check first if the image is set at node level, then try with the default
68        let image = node_config.image().or(chain_context.default_image).cloned();
69
70        // Check first if the command is set at node level, then try with the default
71        let command = if let Some(cmd) = node_config.command() {
72            cmd.clone()
73        } else if let Some(cmd) = chain_context.default_command {
74            cmd.clone()
75        } else {
76            return Err(OrchestratorError::InvalidNodeConfig(
77                node_config.name().into(),
78                "command".to_string(),
79            ));
80        };
81
82        let subcommand = node_config.subcommand().cloned();
83
84        // If `args` is set at `node` level use them
85        // otherwise use the default_args (can be empty).
86        let args: Vec<Arg> = if node_config.args().is_empty() {
87            chain_context
88                .default_args
89                .iter()
90                .map(|x| x.to_owned().clone())
91                .collect()
92        } else {
93            node_config.args().into_iter().cloned().collect()
94        };
95
96        let mut name = node_config.name().to_string();
97        let seed = format!("{}{name}", name.remove(0).to_uppercase());
98        let accounts = generators::generate_jam_node_keys(&seed)?;
99        let accounts = NodeAccounts {
100            seed: seed.clone(),
101            accounts,
102        };
103        let ed25519 = accounts
104            .accounts
105            .get("ed25519")
106            .expect("ed25519 key should be present.");
107        let ed25519_b: [u8; 32] = hex::from_hex(&ed25519.public_key)
108            .expect("ed25519 should be valid")
109            .try_into()
110            .expect("ed25519 should be valid and convert to [u8;32]");
111        let ed25519_pub = ed25519::Public::from(ed25519_b);
112        let peer_id = PeerId(ed25519_pub);
113
114        Ok(Self {
115            name: node_config.name().to_string(),
116            mode: node_config.mode().clone(),
117            peer_id: peer_id.to_string(),
118            image,
119            command,
120            subcommand,
121            args,
122            env: node_config.env().into_iter().cloned().collect(),
123            resources: node_config.resources().cloned(),
124            accounts,
125            port: generators::generate_node_port(None)?,
126            rpc_port: generators::generate_node_port(node_config.rpc_port())?,
127            telemetry_endpoint: node_config.telemetry_endpoint().map(str::to_string),
128        })
129    }
130
131    pub fn command(&self) -> &str {
132        self.command.as_str()
133    }
134
135    pub fn peer_id(&self) -> &str {
136        self.peer_id.as_str()
137    }
138}
