22/// How the chain spec of a JAM chain is produced.
23#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
24pub enum JamChainSpecSource {
25    /// A pre-existing JIP-4 spec, used as given.
26    Path(AssetLocation),
27    /// An external `<cmd> gen-spec <config> <spec>`, opted into with `chain_spec_command` or
28    /// `genesis_overrides`.
29    Generator { command: String },
30    /// Built in process by `jam-chainspec`. The default.
31    JamBuilder(Box<JamGenesisConfig>),
32}
33

65impl JamchainSpec {
66    pub fn from_config(config: &JamchainConfig) -> Result<JamchainSpec, OrchestratorError> {
67        // main command to use, in order:
68        // set as `default_command` or
69        // use the command of the first node.
70        // If non of those is set, return an error.
71        let main_cmd = config
72            .default_command()
73            .or(config.nodes().first().and_then(|node| node.command()))
74            .ok_or(OrchestratorError::InvalidConfig(
75                "Relaychain, either default_command or first node with a command needs to be set."
76                    .to_string(),
77            ))?;
78
79        if config
80            .genesis_overrides()
81            .is_some_and(|overrides| !overrides.is_object())
82        {
83            return Err(OrchestratorError::InvalidConfig(
84                "Jamchain, genesis_overrides must be a JSON object.".to_string(),
85            ));
86        }
87
88        let chain_spec_source = Self::chain_spec_source(config, main_cmd)?;
89
90        // TODO: support podman/docker/k8s
91        // let main_image = config
92        //     .default_image()
93        //     .or(config.nodes().first().and_then(|node| node.image()))
94        //     .map(|image| image.as_str().to_string());
95
96        // build the `node_specs`
97        let chain_context = ChainDefaultContext {
98            default_command: config.default_command(),
99            default_image: config.default_image(),
100            default_resources: config.default_resources(),
101            default_db_snapshot: None,
102            default_args: config.default_args(),
103        };
104
105        let nodes: Vec<JamNodeConfig> = config.nodes().into_iter().cloned().collect();
106        // nodes.extend(
107        //     config
108        //         .group_node_configs()
109        //         .into_iter()
110        //         .flat_map(|node_group| node_group.expand_group_configs()),
111        // );
112
113        let mut names = HashSet::new();
114        let (nodes, mut errs) = nodes
115            .iter()
116            .map(|node_config| JamNodeSpec::from_config(node_config, &chain_context))
117            .fold((vec![], vec![]), |(mut nodes, mut errs), result| {
118                match result {
119                    Ok(mut node) => {
120                        let unique_name =
121                            generate_unique_node_name_from_names(node.name, &mut names);
122                        node.name = unique_name;
123                        nodes.push(node);
124                    },
125                    Err(err) => errs.push(err),
126                }
127                (nodes, errs)
128            });
129
130        if !errs.is_empty() {
131            // TODO: merge errs, maybe return something like Result<Sometype, Vec<OrchestratorError>>
132            return Err(errs.swap_remove(0));
133        }
134
135        Ok(JamchainSpec {
136            id: config.id().clone(),
137            default_command: config.default_command().cloned(),
138            default_image: config.default_image().cloned(),
139            default_resources: config.default_resources().cloned(),
140            default_args: config.default_args().into_iter().cloned().collect(),
141            protocol_params_type: config.protocol_params_type().clone(),
142            chain_spec_source,
143            nodes,
144            genesis_overrides: config.genesis_overrides().cloned(),
145        })
146    }

148    /// Pick how the spec is produced: a given spec, an external generator when one is asked
149    /// for, or else the built-in builder.
150    fn chain_spec_source(
151        config: &JamchainConfig,
152        main_cmd: &Command,
153    ) -> Result<JamChainSpecSource, OrchestratorError> {
154        let generator_field = if config.chain_spec_command().is_some() {
155            Some("chain_spec_command")
156        } else if config.genesis_overrides().is_some() {
157            Some("genesis_overrides")
158        } else {
159            None
160        };
161
162        // The configuration crate refuses these too, this keeps hand-built configs honest.
163        if let (Some(field), Some(_)) = (generator_field, config.genesis()) {
164            return Err(OrchestratorError::InvalidConfig(format!(
165                "Jamchain, `genesis` can't be used with `{field}`."
166            )));
167        }
168
169        if let Some(path) = config.chain_spec_path() {
170            if config.genesis().is_some() || generator_field.is_some() {
171                return Err(OrchestratorError::InvalidConfig(
172                    "Jamchain, `chain_spec_path` can't be used with `genesis`, \
173                     `chain_spec_command` or `genesis_overrides`."
174                        .to_string(),
175                ));
176            }
177            return Ok(JamChainSpecSource::Path(path.clone()));
178        }
179
180        if generator_field.is_some() {
181            let replacements = HashMap::from([
182                ("mainCommand", main_cmd.as_str()),
183                ("subCommand", "gen-spec"),
184            ]);
185            let tmpl = config
186                .chain_spec_command()
187                .map(|cmd| cmd.as_str())
188                .unwrap_or(DEFAULT_JAM_CHAIN_SPEC_TPL_COMMAND);
189            return Ok(JamChainSpecSource::Generator {
190                command: apply_replacements(tmpl, &replacements),
191            });
192        }
193
194        Ok(JamChainSpecSource::JamBuilder(Box::new(
195            config.genesis().cloned().unwrap_or_default(),
196        )))
197    }

214    /// The chain spec is not always generated by the binary that runs the nodes: the two are
215    /// separate commands here, and only the first token of the template is used because the
216    /// orchestrator appends `gen-spec <config> <spec>` itself.
217    #[test]
218    fn a_chain_spec_command_of_its_own_overrides_the_node_binary() {
219        let config = JamchainConfigBuilder::new(Default::default())
220            .with_id("dev")
221            .with_default_command("/bin/node-binary")
222            .with_chain_spec_command("/bin/spec-binary")
223            .with_validator(|node| node.with_name("jam0"))
224            .build()
225            .expect("the chain config builds");
226
227        let spec = JamchainSpec::from_config(&config).expect("the spec builds");
228
229        assert_eq!(
230            spec.chain_spec_source,
231            JamChainSpecSource::Generator {
232                command: "/bin/spec-binary".into()
233            }
234        );
235    }

237    /// The override is merged key by key into the generated config, so anything but an object
238    /// has nothing to merge into — and silently writing the config without it would surface
239    /// only as a chain missing its services.
240    #[test]
241    fn a_genesis_override_that_is_not_an_object_is_refused() {
242        let config = JamchainConfigBuilder::new(Default::default())
243            .with_id("dev")
244            .with_default_command("/bin/node-binary")
245            .with_genesis_overrides(serde_json::json!(["services"]))
246            .with_validator(|node| node.with_name("jam0"))
247            .build()
248            .expect("the chain config builds");
249
250        let error = JamchainSpec::from_config(&config).expect_err("the spec is refused");
251
252        assert!(error.to_string().contains("genesis_overrides"), "{error}");
253    }
