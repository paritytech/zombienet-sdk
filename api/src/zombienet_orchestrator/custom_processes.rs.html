1//! Starting the custom processes of a network once the nodes are up: each one
2//! as soon as its dependencies have met their conditions, in parallel with
3//! whatever else can start, `spawn_concurrency` at a time. A process that
4//! does not meet its condition, or never starts, fails the spawn.
5

21/// Whether a process may start, given what its dependencies reached.
22enum Startable {
23    Now,
24    Later,
25    Never(String),
26}
27
28fn startable(
29    process: &CustomProcess,
30    by_name: &HashMap<&str, &CustomProcess>,
31    states: &HashMap<String, CustomProcessState>,
32) -> Startable {
33    for dep in process.depends_on() {
34        let Some(target) = by_name.get(dep.name.as_str()) else {
35            return Startable::Never(format!(
36                "depends on '{}', which is not a custom process",
37                dep.name
38            ));
39        };
40        let condition = dep.condition_on(target);
41        let Some(state) = states.get(dep.name.as_str()) else {
42            return Startable::Later;
43        };
44        let met = match (condition, state) {
45            (DependencyCondition::ServiceStarted, _) => true,
46            (DependencyCondition::ServiceHealthy, CustomProcessState::Ready) => true,
47            (DependencyCondition::ServiceCompletedSuccessfully, CustomProcessState::Completed) => {
48                true
49            },
50            (_, CustomProcessState::Starting) => return Startable::Later,
51            _ => false,
52        };
53        if !met {
54            return Startable::Never(format!(
55                "depends on '{}' being {}, which it is not ({state})",
56                dep.name,
57                condition.as_str(),
58            ));
59        }
60    }
61    Startable::Now
62}
63
64fn failed(process: &CustomProcess, reason: String) -> OrchestratorError {
65    OrchestratorError::CustomProcessFailed {
66        name: process.name().to_string(),
67        reason,
68    }
69}

71/// Start each process once its dependencies are met, at most
72/// `spawn_concurrency` spawning at a time. The first process that fails, or
73/// is never started, fails the spawn.
74pub(crate) async fn spawn_custom_processes<T: FileSystem>(
75    network: &mut Network<T>,
76    spec: &NetworkSpec,
77    ns: Arc<dyn ProviderNamespace + Send + Sync>,
78    spawn_concurrency: usize,
79) -> Result<(), OrchestratorError> {
80    if spec.custom_processes.is_empty() {
81        return Ok(());
82    }
83    let host_ip = spec
84        .global_settings
85        .local_ip()
86        .copied()
87        .unwrap_or(LOCALHOST);
88    let default_timeout = Duration::from_secs(u64::from(spec.global_settings.node_spawn_timeout()));
89    let mut by_name: HashMap<&str, &CustomProcess> = HashMap::new();
90    let mut states: HashMap<String, CustomProcessState> = HashMap::new();
91
92    // The loader and the builder refuse a name a node or process already has,
93    // but a `NetworkConfig` or `NetworkSpec` deserialized directly skips them,
94    // and a process registered under a node's name would replace the node.
95    // Checked before spawning, so no process runs untracked. Every process
96    // parks its ports before any spawns, see `PreparedProcess`.
97    let mut pending: Vec<PreparedProcess> = vec![];
98    for cp in &spec.custom_processes {
99        if network.has_member(cp.name()) || by_name.insert(cp.name(), cp).is_some() {
100            return Err(failed(
101                cp,
102                "the name is already taken by a node or custom process".into(),
103            ));
104        }
105        let prepared = prepare_process(cp, ns.as_ref())
106            .map_err(|e| failed(cp, format!("could not reserve its ports: {e}")))?;
107        pending.push(prepared);
108    }
109
110    // Every process's ports are known now, so a process may name another's,
111    // `{{ZOMBIE:<process>:port_<name>}}`, the way it names its own.
112    let mut nodes_by_name = network.nodes_json()?;
113    for prepared in &pending {
114        nodes_by_name[prepared.spec.name()] = ports_json(&prepared.ports);
115    }
116
117    // Spawning and waiting are two steps, so a `service_started` dependent
118    // goes as soon as its dependency spawned, while that one is still waited
119    // on. The concurrency limit is on the spawning step, as for nodes.
120    let mut spawning = FuturesUnordered::new();
121    let mut waiting = FuturesUnordered::new();
122    loop {
123        let mut i = 0;
124        while i < pending.len() {
125            match startable(pending[i].spec, &by_name, &states) {
126                Startable::Now if spawning.len() >= spawn_concurrency => i += 1,
127                Startable::Now => {
128                    let prepared = pending.swap_remove(i);
129                    let spec = prepared.spec;
130                    let ns = ns.clone();
131                    let nodes_by_name = &nodes_by_name;
132                    spawning.push(async move {
133                        let outcome = spawn_process(prepared, ns, host_ip, nodes_by_name).await;
134                        (spec, outcome)
135                    });
136                },
137                Startable::Later => i += 1,
138                Startable::Never(reason) => return Err(failed(pending[i].spec, reason)),
139            }
140        }
141        if spawning.is_empty() && waiting.is_empty() {
142            break;
143        }
144        tokio::select! {
145            Some((spec, outcome)) = spawning.next(), if !spawning.is_empty() => {
146                let node = outcome.map_err(|e| failed(spec, format!("{e:#}")))?;
147                states.insert(spec.name().to_string(), CustomProcessState::Starting);
148                let timeout = spec
149                    .timeout()
150                    .map(|secs| Duration::from_secs(u64::from(secs)))
151                    .unwrap_or(default_timeout);
152                waiting.push(async move {
153                    let state = node.wait_ready(timeout).await;
154                    (node, state)
155                });
156            },
157            Some((mut node, state)) = waiting.next(), if !waiting.is_empty() => {
158                if let CustomProcessState::Failed { reason } = state {
159                    return Err(failed(node.spec(), reason));
160                }
161                info!("✅ {}: {state}", node.name());
162                states.insert(node.name().to_string(), state.clone());
163                node.state = state;
164                network.add_running_custom_process(node);
165            },
166        }
167    }
168    // Only a cycle leaves something pending; the config check should have
169    // caught it, so this is defensive.
170    if let Some(prepared) = pending.first() {
171        return Err(failed(
172            prepared.spec,
173            "its dependencies never started (cycle?)".into(),
174        ));
175    }
176
177    Ok(())
178}

186    #[test]
187    fn startable_follows_the_dependency_conditions() {
188        let mk = |name: &str,
189                  deps: Vec<(&str, Option<DependencyCondition>)>,
190                  one_shot: bool,
191                  port: bool| {
192            let mut b = CustomProcessBuilder::new()
193                .with_name(name)
194                .with_command("sh");
195            for (dep, cond) in deps {
196                b = b.with_dependency_on(dep, cond);
197            }
198            if one_shot {
199                b = b.with_one_shot();
200            }
201            if port {
202                b = b.with_named_port("p", 0);
203            }
204            b.build().unwrap()
205        };
206        let db = mk("db", vec![], false, true);
207        let init = mk("init", vec![], true, false);
208        let api = mk("api", vec![("db", None), ("init", None)], false, true);
209        let eager = mk(
210            "eager",
211            vec![("db", Some(DependencyCondition::ServiceStarted))],
212            false,
213            false,
214        );
215        let mismatch = mk(
216            "mismatch",
217            vec![("init", Some(DependencyCondition::ServiceHealthy))],
218            false,
219            false,
220        );
221        let orphan = mk("orphan", vec![("ghost", None)], false, false);
222        let all = [&db, &init, &api, &eager, &mismatch, &orphan];
223        let by_name: HashMap<&str, &CustomProcess> = all.iter().map(|p| (p.name(), *p)).collect();
224        let states =
225            |pairs: Vec<(&str, CustomProcessState)>| -> HashMap<String, CustomProcessState> {
226                pairs.into_iter().map(|(n, s)| (n.to_string(), s)).collect()
227            };
228
229        // nothing known yet: wait
230        assert!(matches!(
231            startable(&api, &by_name, &states(vec![])),
232            Startable::Later
233        ));
234        // db started but not ready, init done: healthy still pending
235        assert!(matches!(
236            startable(
237                &api,
238                &by_name,
239                &states(vec![
240                    ("db", CustomProcessState::Starting),
241                    ("init", CustomProcessState::Completed)
242                ])
243            ),
244            Startable::Later
245        ));
246        // both met
247        assert!(matches!(
248            startable(
249                &api,
250                &by_name,
251                &states(vec![
252                    ("db", CustomProcessState::Ready),
253                    ("init", CustomProcessState::Completed)
254                ])
255            ),
256            Startable::Now
257        ));
258        // service_started is met as soon as the dependency spawned
259        assert!(matches!(
260            startable(
261                &eager,
262                &by_name,
263                &states(vec![("db", CustomProcessState::Starting)])
264            ),
265            Startable::Now
266        ));
267        // a one-shot never becomes healthy
268        assert!(matches!(
269            startable(
270                &mismatch,
271                &by_name,
272                &states(vec![("init", CustomProcessState::Completed)])
273            ),
274            Startable::Never(_)
275        ));
276        // an unknown dependency
277        assert!(matches!(
278            startable(&orphan, &by_name, &states(vec![])),
279            Startable::Never(_)
280        ));
281    }

284    fn a_failure_names_the_process_and_the_reason() {
285        let api = CustomProcessBuilder::new()
286            .with_name("api")
287            .with_command("api")
