79pub async fn spawn_node<'a, T>(
80    node: &NodeSpec,
81    mut files_to_inject: Vec<TransferedFile>,
82    ctx: &SpawnNodeCtx<'a, T>,
83) -> Result<NetworkNode, anyhow::Error>
84where
85    T: FileSystem,
86{
87    let mut created_paths = vec![];
88    // Create and inject the keystore IFF
89    // - The node is validator in the relaychain
90    // - The node is collator (encoded as validator) and the parachain is cumulus_based
91    // (parachain_id) should be set then.
92    if node.is_validator && (ctx.parachain.is_none() || ctx.parachain_id.is_some()) {
93        // Generate keystore for node
94        let node_files_path = if let Some(para) = ctx.parachain {
95            para.id.to_string()
96        } else {
97            node.name.clone()
98        };
99        let asset_hub_polkadot = ctx
100            .parachain_id
101            .map(|id| id.starts_with("asset-hub-polkadot"))
102            .unwrap_or_default();
103        let keystore_key_types = node.keystore_key_types.iter().map(String::as_str).collect();
104        let key_filenames = generators::generate_node_keystore(
105            &node.accounts,
106            &node_files_path,
107            ctx.scoped_fs,
108            asset_hub_polkadot,
109            keystore_key_types,
110        )
111        .await
112        .unwrap();
113
114        // Paths returned are relative to the base dir, we need to convert into
115        // fullpaths to inject them in the nodes.
116        let remote_keystore_chain_id = if let Some(id) = ctx.parachain_id {
117            id
118        } else {
119            ctx.chain_id
120        };
121
122        let keystore_path = node.keystore_path.clone().unwrap_or(PathBuf::from(format!(
123            "/data/chains/{remote_keystore_chain_id}/keystore",
124        )));
125
126        for key_filename in key_filenames {
127            let f = TransferedFile::new(
128                PathBuf::from(format!(
129                    "{}/{}/{}",
130                    ctx.ns.base_dir().to_string_lossy(),
131                    node_files_path,
132                    key_filename.to_string_lossy()
133                )),
134                keystore_path.join(key_filename),
135            );
136            files_to_inject.push(f);
137        }
138        created_paths.push(keystore_path);
139    }
140
141    let base_dir = format!("{}/{}", ctx.ns.base_dir().to_string_lossy(), node.name);
142
143    let (cfg_path, data_path, relay_data_path) = if !ctx.ns.capabilities().prefix_with_full_path {
144        (
145            NODE_CONFIG_DIR.into(),
146            NODE_DATA_DIR.into(),
147            NODE_RELAY_DATA_DIR.into(),
148        )
149    } else {
150        let cfg_path = format!("{}{NODE_CONFIG_DIR}", base_dir);
151        let data_path = format!("{}{NODE_DATA_DIR}", base_dir);
152        let relay_data_path = format!("{}{NODE_RELAY_DATA_DIR}", base_dir);
153        (cfg_path, data_path, relay_data_path)
154    };
155
156    let gen_opts = generators::GenCmdOptions {
157        relay_chain_name: ctx.chain.to_string(),
158        cfg_path,           // TODO: get from provider/ns
159        data_path,          // TODO: get from provider
160        relay_data_path,    // TODO: get from provider
161        use_wrapper: false, // TODO: get from provider
162        bootnode_addr: ctx.bootnodes_addr.clone(),
163        use_default_ports_in_cmd: ctx.ns.capabilities().use_default_ports_in_cmd,
164        // IFF the provider require an image (e.g k8s) we know this is not native
165        is_native: !ctx.ns.capabilities().requires_image,
166    };
167
168    let mut collator_full_node_prom_port: Option<u16> = None;
169    let mut collator_full_node_prom_port_external: Option<u16> = None;
170
171    let (program, args) = match ctx.role {
172        // Collator should be `non-cumulus` one (e.g adder/undying)
173        ZombieRole::Node | ZombieRole::Collator => {
174            let maybe_para_id = ctx.parachain.map(|para| para.id);
175
176            generators::generate_node_command(node, gen_opts.clone(), maybe_para_id)
177        },
178        ZombieRole::CumulusCollator => {
179            let para = ctx.parachain.expect(&format!(
180                "parachain must be part of the context {THIS_IS_A_BUG}"
181            ));
182            collator_full_node_prom_port = node.full_node_prometheus_port.as_ref().map(|p| p.0);
183
184            generators::generate_node_command_cumulus(node, gen_opts.clone(), para.id)
185        },
186        _ => unreachable!(), /* TODO: do we need those?
187                              * ZombieRole::Bootnode => todo!(), */
188    };
189
190    // apply running networ replacements
191    let args: Vec<String> = args
192        .iter()
193        .map(|arg| apply_running_network_replacements(arg, &ctx.nodes_by_name))
194        .collect();
195
196    info!(
197        "🚀 {}, spawning.... with command: {} {}",
198        node.name,
199        program,
200        args.join(" ")
201    );
202
203    let ports = if ctx.ns.capabilities().use_default_ports_in_cmd {
204        // should use default ports to as internal
205        [
206            (P2P_PORT, node.p2p_port.0),
207            (RPC_PORT, node.rpc_port.0),
208            (PROMETHEUS_PORT, node.prometheus_port.0),
209        ]
210    } else {
211        [
212            (P2P_PORT, P2P_PORT),
213            (RPC_PORT, RPC_PORT),
214            (PROMETHEUS_PORT, PROMETHEUS_PORT),
215        ]
216    };
217
218    let resolved_db_snapshot = node
219        .db_snapshot
220        .as_ref()
221        .and_then(|loc| ctx.resolved_db_snapshots.get(loc).cloned());
222    if node.db_snapshot.is_some() && resolved_db_snapshot.is_none() {
223        // Invariant: every NodeSpec.db_snapshot must have been resolved
224        // by orchestrator::generators::resolve_db_snapshots before we
225        // get here. Hitting this path means the resolution step was
226        // skipped for this node.
227        return Err(anyhow::anyhow!(
228            "{THIS_IS_A_BUG}: node {} has db_snapshot {:?} but no resolved cache entry",
229            node.name,
230            node.db_snapshot,
231        ));
232    }
233
234    let spawn_ops = SpawnNodeOptions::new(node.name.clone(), program)
235        .args(args)
236        .env(
237            node.env
238                .iter()
239                .map(|var| (var.name.clone(), var.value.clone())),
240        )
241        .injected_files(files_to_inject)
242        .created_paths(created_paths)
243        .db_snapshot(resolved_db_snapshot)
244        .port_mapping(HashMap::from(ports))
245        .ports(NODE_SERVICE_PORTS)
246        .node_log_path(node.node_log_path.clone())
247        .role(ctx.role.node_role());
248
249    let spawn_ops = if let Some(image) = node.image.as_ref() {
250        spawn_ops.image(image.as_str())
251    } else {
252        spawn_ops
253    };

256        spawn_ops.resources(resources.clone())
257    } else {
258        spawn_ops
259    };
260
261    // Drops the port parking listeners before spawn
262    node.ws_port.drop_listener();
263    node.p2p_port.drop_listener();
264    node.rpc_port.drop_listener();
265    node.prometheus_port.drop_listener();

267        port.drop_listener();
268    }

270        port.drop_listener();
271    }
272
273    let running_node = ctx.ns.spawn_node(&spawn_ops).await.with_context(|| {
274        format!(
275            "Failed to spawn node: {} with opts: {:#?}",
276            node.name, spawn_ops
277        )
278    })?;
279
280    let mut ip_to_use = if let Some(local_ip) = ctx.global_settings.local_ip() {
281        *local_ip
282    } else {
283        LOCALHOST
284    };
285
286    let (rpc_port_external, prometheus_port_external, p2p_external);
287
288    if running_in_ci() && ctx.ns.provider_name() == "k8s" {
289        // running kubernets in ci require to use ip and default port
290        (rpc_port_external, prometheus_port_external, p2p_external) =
291            (RPC_PORT, PROMETHEUS_PORT, P2P_PORT);
292        collator_full_node_prom_port_external = Some(FULL_NODE_PROMETHEUS_PORT);
293        ip_to_use = running_node.ip().await?;
294    } else {

296        let ports = futures::future::try_join_all(vec![
297            running_node.create_port_forward(node.rpc_port.0, RPC_PORT),
298            running_node.create_port_forward(node.prometheus_port.0, PROMETHEUS_PORT),
299        ])
300        .await?;
301
302        (rpc_port_external, prometheus_port_external, p2p_external) = (

304            ports[1].unwrap_or(node.prometheus_port.0),
305            // p2p don't need port-fwd
306            node.p2p_port.0,
307        );
308
309        if let Some(full_node_prom_port) = collator_full_node_prom_port {
310            let port_fwd = running_node
311                .create_port_forward(full_node_prom_port, FULL_NODE_PROMETHEUS_PORT)
312                .await?;
313            collator_full_node_prom_port_external = Some(port_fwd.unwrap_or(full_node_prom_port));
314        }
315    }
316
317    let running_ip = running_node.ip().await?;
318
319    let multiaddr = generators::generate_node_bootnode_addr(
320        &node.peer_id,
321        &running_ip,
322        if ctx.ns.provider_name() == "k8s" {
323            P2P_PORT
324        } else {
325            p2p_external
326        }, // for k8s use always the internal port
327        running_node.args().as_ref(),
328        &node.p2p_cert_hash,
329    )?;
330
331    let ws_uri = format!("ws://{ip_to_use}:{rpc_port_external}");
332    let internal_ws_uri = if ctx.ns.provider_name() == "k8s" {
333        format!("ws://{running_ip}:{RPC_PORT}")
334    } else {
335        // docker/native can use the same
336        ws_uri.clone()
337    };
338    let prometheus_uri = format!("http://{ip_to_use}:{prometheus_port_external}/metrics");
339    info!("🚀 {}, should be running now", node.name);
340    info!(
341        "💻 {}: direct link (pjs) https://polkadot.js.org/apps/?rpc={ws_uri}#/explorer",
342        node.name
343    );
344    info!(
345        "💻 {}: direct link (papi) https://dev.papi.how/explorer#networkId=custom&endpoint={ws_uri}",
346        node.name
347    );
348
349    info!("📊 {}: metrics link {prometheus_uri}", node.name);
350
351    if let Some(full_node_prom_port) = collator_full_node_prom_port_external {
352        info!(
353            "📊 {}: collator full-node metrics link http://{}:{}/metrics",
354            node.name, ip_to_use, full_node_prom_port
355        );
356    }
357
358    info!("📓 logs cmd: {}", running_node.log_cmd());
359
360    let node_ctx = if let Some(parachain) = ctx.parachain {
361        NodeContext::Para {
362            para_id: parachain.id,
363            is_cumulus_based: parachain.is_cumulus_based,
364        }
365    } else {
366        NodeContext::Rc
367    };
368
369    Ok(NetworkNode::new(
370        node.name.clone(),
371        ws_uri,
372        internal_ws_uri,
373        prometheus_uri,
374        multiaddr,
375        node.clone(),
376        running_node,
377        gen_opts,
378        node_ctx,
379    ))
380}

382/// The `args` and `env` of a custom process, placeholders resolved.
383type ResolvedArgsAndEnv = (Vec<String>, Vec<(String, String)>);
384
385/// The args and env of `custom_process` with every `{{ZOMBIE:<name>:<field>}}`
386/// resolved against the nodes already running (`nodes_by_name`, as the
387/// orchestrator keeps them) and the process's own ports, as `port_<name>`.
388///
389/// A placeholder that stays unresolved is an error: the process would start
390/// with the literal text.
391fn resolve_placeholders(
392    custom_process: &CustomProcess,
393    resolved_ports: &[(String, Port)],
394    nodes_by_name: &serde_json::Value,
395) -> Result<ResolvedArgsAndEnv, anyhow::Error> {
396    let name = custom_process.name();
397    let mut context = nodes_by_name.clone();
398    context[name] = serde_json::json!(resolved_ports
399        .iter()
400        .map(|(port_name, port)| (format!("port_{port_name}"), port.to_string()))
401        .collect::<HashMap<_, _>>());
402
403    let resolve = |text: &str| -> Result<String, anyhow::Error> {
404        let resolved = apply_running_network_replacements(text, &context);
405        if has_tokens(&resolved) {
406            Err(anyhow::anyhow!(
407                "{name}: unresolved placeholder in {resolved:?} (nodes in context: {})",
408                context
409                    .as_object()
410                    .map(|nodes| nodes.keys().cloned().collect::<Vec<_>>().join(", "))
411                    .unwrap_or_default()
412            ))
413        } else {
414            Ok(resolved)
415        }
416    };
417
418    let args = custom_process
419        .args()
420        .iter()
421        .flat_map(|arg| arg.to_vec())
422        .map(|arg| resolve(&arg))
423        .collect::<Result<Vec<_>, _>>()?;
424    let env = custom_process
425        .env()
426        .iter()
427        .map(|var| Ok((var.name.clone(), resolve(&var.value)?)))
428        .collect::<Result<Vec<_>, anyhow::Error>>()?;
429
430    Ok((args, env))
431}

433/// A custom process with its ports parked, ready to spawn.
434///
435/// Parking happens for every process before any of them spawns, so a port one
436/// process picks (or declares, on native) cannot be handed to another; each
437/// process releases its own right before its spawn.
438pub(crate) struct PreparedProcess<'a> {
439    spec: &'a CustomProcess,
440    /// Every declared port by name, `0`s resolved to the picked number.
441    ports: Vec<(String, Port)>,
442    /// On docker, the host port each port is published on.
443    host_ports: HashMap<Port, Port>,
444    parked: Vec<ParkedPort>,
445}
446
447/// Resolve and park the ports of `custom_process`: a `0` is picked free, a
448/// fixed one on native has to be free too (or the process loses the bind and
449/// whoever holds the port answers for it), and on docker each port gets a
450/// parked host port to be published on, so two networks can share a host.
451pub(crate) fn prepare_process<'a>(
452    custom_process: &'a CustomProcess,
453    ns: &dyn ProviderNamespace,
454) -> Result<PreparedProcess<'a>, anyhow::Error> {
455    let name = custom_process.name();
456    let mut ports: Vec<(String, Port)> = Vec::with_capacity(custom_process.ports().len());
457    let mut host_ports: HashMap<Port, Port> = HashMap::new();
458    let mut parked = vec![];
459    // A picked port may not land on a fixed one of the same process.
460    let fixed: Vec<Port> = custom_process
461        .ports()
462        .iter()
463        .map(|p| p.port)
464        .filter(|p| *p != 0)
465        .collect();
466    for declared in custom_process.ports() {
467        let port = if declared.port == 0 {
468            let picked = loop {
469                let picked = generators::generate_node_port(None)?;
470                if !fixed.contains(&picked.0) {
471                    break picked;
472                }
473            };
474            let port = picked.0;
475            parked.push(picked);
476            port
477        } else if !ns.capabilities().requires_image {
478            let picked =
479                generators::generate_node_port(Some(declared.port)).with_context(|| {
480                    format!(
481                        "{name}: port {} ('{}') is already in use",
482                        declared.port, declared.name
483                    )
484                })?;
485            parked.push(picked);
486            declared.port
487        } else {
488            declared.port
489        };
490        if ns.provider_name() == "docker" {
491            let host = generators::generate_node_port(None)?;
492            host_ports.insert(port, host.0);
493            parked.push(host);
494        }
495        ports.push((declared.name.clone(), port));
496    }
497
498    Ok(PreparedProcess {
499        spec: custom_process,
500        ports,
501        host_ports,
502        parked,
503    })
504}

506/// Spawn a custom process and make it a member of the network: the running
507/// handle is returned with every declared port resolved to where it is
508/// reachable, so it can be tracked, written to `zombie.json` and reattached.
509///
510/// The process runs without zombienet's wrapper script: it is an
511/// off-the-shelf image most of the time, which may lack `bash`, and the
512/// pause/resume controls the wrapper provides are for nodes.
513///
514/// Every port is offered to the process as `{{ZOMBIE:<process>:port_<name>}}`
515/// in its args and env.
516pub(crate) async fn spawn_process(
517    prepared: PreparedProcess<'_>,
518    ns: Arc<dyn ProviderNamespace + Send + Sync>,
519    host_ip: IpAddr,
520    nodes_by_name: &serde_json::Value,
521) -> Result<CustomProcessNode, anyhow::Error> {
522    let PreparedProcess {
523        spec: custom_process,
524        ports: resolved_ports,
525        host_ports,
526        parked,
527    } = prepared;
528    let provider = ns.provider_name();
529    let capabilities = ns.capabilities();
530    let name = custom_process.name();
531
532    let (args, env) = resolve_placeholders(custom_process, &resolved_ports, nodes_by_name)?;
533
534    let spawn_ops = SpawnNodeOptions::new(name, custom_process.command().as_str())
535        .args(&args)
536        .env(env)
537        .ports(resolved_ports.iter().map(|(n, p)| (n.as_str(), *p)))
538        // only docker publishes ports; k8s and native ignore the mapping
539        .port_mapping(host_ports.clone())
540        .role(NodeRole::CustomProcess)
541        .without_wrapper();
542
543    let spawn_ops = if let Some(image) = custom_process.image() {
544        spawn_ops.image(image.as_str())
545    } else {
546        spawn_ops
547    };
548
549    let spawn_ops = match custom_process.resources() {
550        Some(resources) if capabilities.has_resources => spawn_ops.resources(resources.clone()),
551        Some(_) => {
552            warn!(
553                "⚠️  {name}: resources are not supported by the {provider} provider, ignoring them"
554            );
555            spawn_ops
556        },
557        None => spawn_ops,
558    };
559
560    info!(
561        "🚀 {name}, spawning custom process.... with command: {} {}",
562        custom_process.command().as_str(),
563        args.join(" ")
564    );
565
566    for port in &parked {
567        port.drop_listener();
568    }
569
570    let running_node = ns
571        .spawn_node(&spawn_ops)
572        .await
573        .with_context(|| format!("Failed to spawn node: {name} with opts: {:#?}", spawn_ops))?;
574
575    // From here on the process runs. Nothing below may lose it: a port we
576    // cannot reach from here is reported by its in-network address instead.
577    let running_ip = match running_node.ip().await {
578        Ok(ip) => ip,
579        Err(err) => {
580            warn!("⚠️  {name}: could not get the process ip ({err}); using {host_ip}");
581            host_ip
582        },
583    };
584
585    let in_ci = running_in_ci();
586    let mut ports = std::collections::BTreeMap::new();
587    for (port_name, port) in &resolved_ports {
588        // `internal` is the address inside the network, `external` the one
589        // from where zombienet runs.
590        let (internal, external) = match provider {
591            // The Service named after the process; the pod itself from inside
592            // the cluster (CI), a port-forward on this host otherwise.
593            "k8s" => {
594                let internal = format!("{name}:{port}");
595                let external = if in_ci {
596                    format!("{running_ip}:{port}")
597                } else {
598                    forward_or_keep(&running_node, name, port_name, *port, internal.clone()).await
599                };
600                (internal, external)
601            },
602            // The container; the port it is published on.
603            "docker" => (
604                format!("{running_ip}:{port}"),
605                format!("{host_ip}:{}", host_ports[port]),
606            ),
607            // The host, both ways.
608            _ => (format!("{host_ip}:{port}"), format!("{host_ip}:{port}")),
609        };
610
611        ports.insert(
612            port_name.clone(),
613            ProcessPort {
614                port: *port,
615                external,
616                internal,
617            },
618        );
619    }
620
621    info!("🚀 {name}, should be running now");
622    for (port_name, port) in &ports {
623        info!(
624            "🔌 {name}: port {port_name} ({}) reachable at {}",
625            port.port, port.external
626        );
627    }
628    info!("📓 logs cmd: {}", running_node.log_cmd());
629
630    Ok(CustomProcessNode::new(
631        name,
632        running_node,
633        custom_process.clone(),
634        running_ip,
635        ports,
636    ))
637}

639/// A port-forward to `port` of `node`, as `host:port` on this host, or
640/// `fallback` where the provider has none to offer or it could not be opened.
641pub(crate) async fn forward_or_keep(
642    node: &DynNode,
643    process: &str,
644    port_name: &str,
645    port: Port,
646    fallback: String,
647) -> String {
648    match node.create_port_forward(0, port).await {
649        Ok(Some(local_port)) => format!("{LOCALHOST}:{local_port}"),
650        Ok(None) => fallback,
651        Err(err) => {
652            warn!("⚠️  {process}: could not forward port {port_name} ({port}): {err}");
653            fallback
654        },
655    }
656}

658/// Spawn the custom processes of `spec`, concurrently, and register the ones
659/// that came up. A process that does not is logged and left out: the nodes
660/// are up by now, and a missing side service must not take the network down.
661pub(crate) async fn spawn_custom_processes<T: FileSystem>(
662    network: &mut Network<T>,
663    spec: &NetworkSpec,
664    ns: Arc<dyn ProviderNamespace + Send + Sync>,
665) -> Result<(), anyhow::Error> {
666    if spec.custom_processes.is_empty() {
667        return Ok(());
668    }
669    let host_ip = spec
670        .global_settings
671        .local_ip()
672        .copied()
673        .unwrap_or(LOCALHOST);
674    let nodes_by_name = network.nodes_json()?;
675
676    // Checked before spawning: a process that ran and was then refused would
677    // be untracked, having taken the node's place in the provider's registry.
678    let (taken, free): (Vec<_>, Vec<_>) = spec
679        .custom_processes
680        .iter()
681        .partition(|cp| network.has_member(cp.name()));
682    for cp in taken {
683        warn!(
684            "⚠️  Custom process {} not spawned: the name is already taken by a node or custom process",
685            cp.name()
686        );
687    }
688
689    // Every process parks its ports before any spawns, see `PreparedProcess`.
690    let prepared: Vec<PreparedProcess> = free
691        .into_iter()
692        .filter_map(|cp| match prepare_process(cp, ns.as_ref()) {
693            Ok(prepared) => Some(prepared),
694            Err(e) => {
695                warn!(
696                    "⚠️  Failed to reserve the ports of custom process {}, not spawned, err: {e}",
697                    cp.name()
698                );
699                None
700            },
701        })
702        .collect();
703    let names: Vec<&str> = prepared.iter().map(|p| p.spec.name()).collect();
704    let spawning = prepared
705        .into_iter()
706        .map(|p| spawn_process(p, ns.clone(), host_ip, &nodes_by_name));
707    for (name, spawned) in names
708        .into_iter()
709        .zip(futures::future::join_all(spawning).await)
710    {
711        match spawned {
712            Ok(process) => network.add_running_custom_process(process),
713            Err(e) => warn!("⚠️  Failed to spawn custom process {name}, err: {e}"),
714        }
715    }
716
717    Ok(())
718}

720pub async fn spawn_jam_node<'a, T>(
721    node: &JamNodeSpec,
722    mut files_to_inject: Vec<TransferedFile>,
723    ctx: &SpawnNodeCtx<'a, T>,
724) -> Result<JamNetworkNode, anyhow::Error>
725where
726    T: FileSystem,
727{
728    let base_dir = format!("{}/{}", ctx.ns.base_dir().to_string_lossy(), node.name);
729
730    let (cfg_path, data_path) = if !ctx.ns.capabilities().prefix_with_full_path {
731        (NODE_CONFIG_DIR.into(), NODE_DATA_DIR.into())
732    } else {
733        let cfg_path = format!("{}{NODE_CONFIG_DIR}", base_dir);
734        let data_path = format!("{}{NODE_DATA_DIR}", base_dir);
735        (cfg_path, data_path)
736    };
737
738    // create local seed file and set to transfer process:
739
740    // 1. Create local dir to store all the needed files
741    ctx.scoped_fs
742        .create_dir_all(PathBuf::from(&node.name))
743        .await?;
744
745    // 2. set paths to use (local and remote)
746    let keys_remote = PathBuf::from(format!("{}/{}/keys", NODE_CONFIG_DIR, ctx.chain));
747    let seed_local_path = PathBuf::from(format!("{}/{}.seed", base_dir, node.name));
748    let seed_remote_path = PathBuf::from(format!(
749        "{}/{}.seed",
750        keys_remote.to_string_lossy(),
751        node.name
752    ));
753
754    // 3. write local seed
755    ctx.scoped_fs
756        .write(&seed_local_path, hash_raw(node.accounts.seed.as_bytes()))
757        .await?;
758
759    // 4. set paths/files to use in remote.
760    let created_paths = vec![keys_remote];
761    files_to_inject.push(TransferedFile::new(seed_local_path, seed_remote_path));
762
763    let gen_opts = generators::GenCmdOptions {
764        relay_chain_name: ctx.chain.to_string(),
765        cfg_path,  // TODO: get from provider/ns
766        data_path, // TODO: get from provider
767        use_default_ports_in_cmd: ctx.ns.capabilities().use_default_ports_in_cmd,
768        // IFF the provider require an image (e.g k8s) we know this is not native
769        is_native: !ctx.ns.capabilities().requires_image,
770        bootnode_addr: ctx.bootnodes_addr.clone(),
771        ..Default::default()
772    };
773
774    let (program, args) = generators::generate_jam_node_command(node, gen_opts.clone());
775    // apply running networ replacements
776    let args: Vec<String> = args
777        .iter()
778        .map(|arg| apply_running_network_replacements(arg, &ctx.nodes_by_name))
779        .collect();
780
781    info!(
782        "🚀 {}, spawning.... with command: {} {}",
783        node.name,
784        program,
785        args.join(" ")
786    );
787
788    let ports = if ctx.ns.capabilities().use_default_ports_in_cmd {
789        // should use default ports to as internal
790        [(JAM_PORT, node.port.0), (RPC_PORT, node.rpc_port.0)]
791    } else {
792        [(JAM_PORT, JAM_PORT), (RPC_PORT, RPC_PORT)]
793    };
794
795    let spawn_ops = SpawnNodeOptions::new(node.name.clone(), program)
796        .args(args)
797        .env(
798            node.env
799                .iter()
800                .map(|var| (var.name.clone(), var.value.clone())),
801        )
802        .injected_files(files_to_inject)
803        .created_paths(created_paths)
804        .port_mapping(HashMap::from(ports))
805        .ports(NODE_SERVICE_PORTS);
806
807    let spawn_ops = if let Some(image) = node.image.as_ref() {
808        spawn_ops.image(image.as_str())
809    } else {
810        spawn_ops
811    };
812
813    let spawn_ops = if let Some(resources) = node.resources.as_ref() {
814        spawn_ops.resources(resources.clone())
815    } else {
816        spawn_ops
817    };
818
819    // Drops the port parking listeners before spawn
820    node.port.drop_listener();
821    node.rpc_port.drop_listener();
822
823    let running_node = ctx.ns.spawn_node(&spawn_ops).await.with_context(|| {
824        format!(
825            "Failed to spawn node: {} with opts: {:#?}",
826            node.name, spawn_ops
827        )
828    })?;
829
830    let ip_to_use = if let Some(local_ip) = ctx.global_settings.local_ip() {
831        *local_ip
832    } else {
833        LOCALHOST
834    };
835
836    // NOTE: running in ci (k8s) is not supported yet for JAM, so we don't
837    // create port-forwards nor use the internal ip/default ports here.
838
839    info!("🚀 {}, should be running now", node.name);
840    match node.mode {
841        JamNodeMode::Ordinary => {
842            info!("💻 {}, rpc  {ip_to_use}:{}", node.name, node.rpc_port.0);
843        },
844        JamNodeMode::Validator | JamNodeMode::Proxy => {
845            info!(
846                "💻 {}, peer details {}@{ip_to_use}:{}",
847                node.name, node.peer_id, node.port.0
848            );
849        },
850    }
851
852    info!("📓 logs cmd: {}", running_node.log_cmd());
853
854    Ok(JamNetworkNode::new(
855        node.name.clone(),
856        running_node,
857        node.clone(),
858        ip_to_use,
859        gen_opts,
860    ))
861}
